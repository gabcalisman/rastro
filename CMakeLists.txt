cmake_minimum_required(VERSION 3.22)
project(Rastro VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
 # Conforming nested lambda captures and consistent PT/EN source encoding.
 add_compile_options(/Zc:lambda /utf-8)
endif()
option(RASTRO_CORE_ONLY "Build only portable DSP tests" OFF)
enable_testing()
add_executable(rastro_tests tests/EngineTests.cpp)
target_include_directories(rastro_tests PRIVATE Source)
add_test(NAME dsp COMMAND rastro_tests)
add_executable(rastro_character_tests tests/CharacterTests.cpp)
target_include_directories(rastro_character_tests PRIVATE Source)
add_test(NAME character COMMAND rastro_character_tests)
add_executable(rastro_playback_tests tests/PlaybackTests.cpp)
target_include_directories(rastro_playback_tests PRIVATE Source)
add_test(NAME playback COMMAND rastro_playback_tests)
add_executable(rastro_mp3_tests tests/Mp3Tests.cpp Source/Mp3Decoder.cpp)
target_include_directories(rastro_mp3_tests PRIVATE Source)
target_compile_definitions(rastro_mp3_tests PRIVATE RASTRO_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME mp3 COMMAND rastro_mp3_tests)
add_executable(rastro_synth_tests tests/SynthTests.cpp)
target_include_directories(rastro_synth_tests PRIVATE Source)
add_test(NAME synth COMMAND rastro_synth_tests)
add_executable(rastro_capture_tests tests/CaptureTests.cpp)
target_include_directories(rastro_capture_tests PRIVATE Source)
add_test(NAME capture COMMAND rastro_capture_tests)
add_executable(rastro_timing_tests tests/TimingTests.cpp)
target_include_directories(rastro_timing_tests PRIVATE Source)
add_test(NAME timing COMMAND rastro_timing_tests)
find_package(Threads REQUIRED)
add_executable(rastro_patch_tests tests/PatchTests.cpp)
target_include_directories(rastro_patch_tests PRIVATE Source)
target_link_libraries(rastro_patch_tests PRIVATE Threads::Threads)
add_test(NAME patches COMMAND rastro_patch_tests)
if(RASTRO_CORE_ONLY)
 return()
endif()
find_package(JUCE 8.0.6 CONFIG QUIET)
if(NOT JUCE_FOUND)
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
endif()
set(formats VST3 Standalone)
if(APPLE)
 list(APPEND formats AU)
endif()
# The Mac build may use the redesigned web editor; the original native JUCE editor remains the fallback.
set(rastro_web_default OFF)
if(APPLE)
    set(rastro_web_default ON)
endif()
option(RASTRO_WEB_UI "Use the redesigned web interface for RASTRO PLAY" ${rastro_web_default})
juce_add_plugin(Rastro COMPANY_NAME "Gab Calisman" BUNDLE_ID com.gabcalisman.rastro PLUGIN_MANUFACTURER_CODE GbCa PLUGIN_CODE Rstr FORMATS ${formats} PRODUCT_NAME "RASTRO" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE NEEDS_WEB_BROWSER ${RASTRO_WEB_UI} COPY_PLUGIN_AFTER_BUILD FALSE MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Record audio into RASTRO samples when you choose Capture.")
target_sources(Rastro PRIVATE Source/WebEditor.cpp Source/Processor.cpp Source/Editor.cpp Source/Performance.cpp Source/Hardware.cpp Source/CapturePanel.cpp Source/InstrumentPanel.cpp Source/PatchPanel.cpp Source/Customizer.cpp Source/RecordingImport.cpp Source/Mp3Decoder.cpp Source/SampleView.cpp)
# Linux M4A uses a separately installed offline executable, never linked or bundled.
set(RASTRO_FFMPEG_EXECUTABLE "/usr/bin/ffmpeg" CACHE FILEPATH "Offline FFmpeg executable for Linux M4A")
target_compile_definitions(Rastro PRIVATE RASTRO_FFMPEG_EXECUTABLE="${RASTRO_FFMPEG_EXECUTABLE}")
if(RASTRO_WEB_UI)
    file(GLOB rastro_web_files CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/ui/*.html ${CMAKE_CURRENT_SOURCE_DIR}/ui/*.js ${CMAKE_CURRENT_SOURCE_DIR}/ui/*.css ${CMAKE_CURRENT_SOURCE_DIR}/ui/*.svg ${CMAKE_CURRENT_SOURCE_DIR}/ui/*.png ${CMAKE_CURRENT_SOURCE_DIR}/ui/*.jpg ${CMAKE_CURRENT_SOURCE_DIR}/ui/*.woff)
    juce_add_binary_data(RastroWebAssets HEADER_NAME RastroWeb.h NAMESPACE RastroWeb SOURCES ${rastro_web_files})
    target_link_libraries(Rastro PRIVATE RastroWebAssets)
    target_compile_definitions(Rastro PUBLIC RASTRO_WEB_UI=1 JUCE_WEB_BROWSER=1)
else()
    target_compile_definitions(Rastro PUBLIC RASTRO_WEB_UI=0 JUCE_WEB_BROWSER=0)
endif()
target_compile_definitions(Rastro PUBLIC JUCE_USE_CUSTOM_PLUGIN_STANDALONE_APP=1 JUCE_USE_CURL=0 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Rastro PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
target_sources(Rastro_Standalone PRIVATE Source/Standalone.cpp)
option(RASTRO_INTEGRATION_TESTS "Build processor and GUI integration checks" OFF)
if(RASTRO_INTEGRATION_TESTS)
 target_compile_definitions(Rastro PUBLIC JUCE_MODAL_LOOPS_PERMITTED=1)
 add_executable(rastro_integration tests/ProcessorTests.cpp)
 target_include_directories(rastro_integration PRIVATE Source "${JUCE_MODULES_DIR}")
 target_compile_definitions(rastro_integration PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 RASTRO_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
 target_link_libraries(rastro_integration PRIVATE Rastro)
 add_test(NAME processor COMMAND rastro_integration)
endif()
option(RASTRO_UI_BENCHMARK "Build local native-window UI measurement harness" OFF)
if(RASTRO_UI_BENCHMARK)
 add_executable(rastro_ui_benchmark tests/UiBenchmark.cpp)
 target_include_directories(rastro_ui_benchmark PRIVATE Source "${JUCE_MODULES_DIR}")
 target_compile_definitions(rastro_ui_benchmark PRIVATE RASTRO_UI_METRICS=1 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 RASTRO_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
 target_link_libraries(rastro_ui_benchmark PRIVATE Rastro)
endif()

# Complementary recorder only: this is not RASTRO FX and has no effects rack.
set(capture_formats VST3)
if(APPLE)
 list(APPEND capture_formats AU)
endif()
juce_add_plugin(RastroCapture COMPANY_NAME "Gab Calisman" BUNDLE_ID com.gabcalisman.rastro.capture PLUGIN_MANUFACTURER_CODE GbCa PLUGIN_CODE Rcpt FORMATS ${capture_formats} PRODUCT_NAME "RASTRO Capture" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(RastroCapture PRIVATE Source/CaptureProcessor.cpp Source/CapturePanel.cpp)
target_compile_definitions(RastroCapture PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(RastroCapture PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)

if(APPLE OR UNIX)
 add_executable(rastro_recording_tests tests/RecordingTests.cpp Source/RecordingImport.cpp)
 target_include_directories(rastro_recording_tests PRIVATE Source)
 target_compile_definitions(rastro_recording_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0 RASTRO_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" RASTRO_FFMPEG_EXECUTABLE="${RASTRO_FFMPEG_EXECUTABLE}")
 target_link_libraries(rastro_recording_tests PRIVATE juce::juce_audio_formats juce::juce_recommended_config_flags)
 add_test(NAME recordings COMMAND rastro_recording_tests)
 add_executable(rastro_capture_bridge_tests tests/CaptureBridgeTests.cpp)
 target_include_directories(rastro_capture_bridge_tests PRIVATE Source)
 target_compile_definitions(rastro_capture_bridge_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0)
 target_link_libraries(rastro_capture_bridge_tests PRIVATE juce::juce_audio_formats juce::juce_recommended_config_flags Threads::Threads)
 add_test(NAME capture_bridge COMMAND rastro_capture_bridge_tests)
endif()
